\section{How Lab Works}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% What is lab
\begin{frame}
	\frametitle{What Is a Lab Section?}
	Lab is \hl{not} a second lecture. No new theory gets delivered here.
	\vfill
	\begin{itemize}
		\item You work in \hl{pairs or small groups}
		\item You use \hl{R and RStudio} on real data sets
		\item You hand in a \hl{worksheet} at the end of every session
		\item I circulate and help; you should be talking to your partner, not to a screen in silence
	\end{itemize}
	\vfill
	\Note{Lecture is where ideas are introduced. Lab is where you practice them with your hands on the keyboard.}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Weekly rhythm
\begin{frame}
	\frametitle{The Weekly Rhythm}
	Every lab follows the same two steps:
	\vfill
	\begin{enumerate}
		\item \hl{Before lab:} complete the assigned \textbf{tutorial} and submit its \textbf{hash code} to Blackboard
		\item \hl{During lab:} work through the \textbf{worksheet} with your partner, and hand it in
		%\item \hl{After lab:} finish any post-lab submission listed on the worksheet
	\end{enumerate}
	\vfill
	\Remember{Today is the one exception. We will do Tutorial 0 together, in lab. From Lab 2 onward the tutorial is done \textit{before} you walk in.}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Tutorials
\begin{frame}
	\frametitle{Tutorials and Hash Codes}
	Tutorials are interactive R lessons: guided explanations, examples, and exercises.
	\vfill
	\begin{itemize}
		\item Each tutorial ends by generating a \hl{hash code}
		\item That hash code is your \hl{proof of completion}
		\item You submit the hash code \hl{to Blackboard}
	\end{itemize}
	\vfill
	\red{The hash code goes to Blackboard. Writing it on your paper worksheet does not count as submitting it.}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

	\frametitle{How Lab Work Is Graded}

		\item Tutorials are credited from the hash code you submit to Blackboard
		\item One submission covers both the hash code and the worksheet for that lab
	\end{itemize}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% R vs RStudio
\begin{frame}
	\frametitle{R and RStudio Are Two Different Things}
	\begin{itemize}
		\item \hl{R} is the language and the engine that does the computing
		\item \hl{RStudio} is the workspace you use to write and run R code
	\end{itemize}
	\vfill
	\red{Install R first, then RStudio.} RStudio is only a front end; it will not run without R underneath it.
	\vfill
	\begin{itemize}
		\item R: \webURL{https://cloud.r-project.org}
		\item RStudio Desktop: \webURL{https://posit.co/download/rstudio-desktop/}
	\end{itemize}
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
% Check install
\begin{frame}[fragile]
	\frametitle{Checking Your Install}
	Open RStudio and type this into the \hl{Console}, then press Enter:
	\vfill
	\begin{semiverbatim}
	R.version.string
	\end{semiverbatim}
	\vfill
	If it prints a version number, you are ready to go.
	\vfill
	If it does not, or RStudio will not open at all, \hl{raise your hand now} rather than waiting.
\end{frame}
