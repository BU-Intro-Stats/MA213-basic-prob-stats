\textbf{Simple Linear Regression}

\textit{Model:}
\[ y_i = \beta_0 + \beta_1 x_i + \epsilon_i \]
where $\epsilon_i \stackrel{iid}{\sim} N(0, \sigma^2)$ 

Or:
\[\hat{y} = b_0 + b_1 x\]

\textit{Estimating the least squares coefficients:}
\begin{align*}
    b_1 &= \frac{s_{y}}{s_{x}} R \\
    b_0 &= \bar{y} - b_1 \bar{x}
\end{align*} 

\textit{Conditions:}
\begin{itemize}
    \item Linearity: The relationship between $x$ and $y$ is linear
    \item Normality: The residuals are independent and approximately normally distributed
    \item Equal Variance: The residuals have constant variance (homoscedasticity)
\end{itemize}

\textit{Inference for slope $\beta_1$:}
\[ \frac{b_1-\beta_1}{SE_{b_1}} \sim t_{df} \qquad df = n-2 \]
